rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Host: boleh dibaca & didaftarkan, tidak boleh diubah/dihapus dari klien
    match /hosts/{username} {
      allow get, create: if true;
      allow list, update, delete: if false;
    }

    // Event + SEMUA subkoleksinya (questions, responses, dll)
    // Catch-all {document=**} inilah yang bikin subkoleksi "responses" (kuesioner) bisa jalan.
    match /events/{event} {
      allow read, write: if true;

      match /{document=**} {
        allow read, write: if true;
      }
    }
  }
}

====================================================================
 CATATAN
 - Baris "match /{document=**}" WAJIB ada. Tanpa itu, submit kuesioner
   gagal dengan error permission-denied (subkoleksi "responses" diblokir).
 - Rules ini terbuka (siapa pun bisa baca/tulis) — sesuai kebutuhan
   event internal tanpa login. Untuk jangka panjang, pertimbangkan
   membatasi per-event atau memakai Firebase Auth.
