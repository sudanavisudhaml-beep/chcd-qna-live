rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Host: boleh dibaca & didaftarkan, tidak boleh diubah/dihapus dari klien
    match /hosts/{username} {
      allow get, create: if true;
      allow list, update, delete: if false;
    }

    // Event + SEMUA subkoleksinya (questions, responses, dll)
    match /events/{event} {
      allow read, write: if true;

      // Baris catch-all inilah yang membuka subkoleksi "responses" (kuesioner).
      // Tanpa ini, submit kuesioner gagal: permission-denied.
      match /{document=**} {
        allow read, write: if true;
      }
    }
  }
}
